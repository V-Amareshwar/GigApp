## MIGRATION 002 — USERS TABLE

```sql
CREATE TABLE users (
  id                  UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  phone               VARCHAR(15) UNIQUE NOT NULL,
  name                VARCHAR(100),
  profile_photo_url   TEXT,
  city                VARCHAR(100),
  latitude            DECIMAL(9,6),
  longitude           DECIMAL(9,6),
  is_phone_verified   BOOLEAN DEFAULT FALSE,
  is_id_verified      BOOLEAN DEFAULT FALSE,
  trust_score         DECIMAL(3,2) DEFAULT 0.00,
  current_role        VARCHAR(20) DEFAULT 'seeker'
                        CHECK (current_role IN ('seeker', 'provider')),
  is_banned           BOOLEAN DEFAULT FALSE,
  ban_reason          TEXT,
  fcm_token           TEXT,                    -- Firebase push token
  created_at          TIMESTAMPTZ DEFAULT NOW(),
  updated_at          TIMESTAMPTZ DEFAULT NOW(),
  last_active_at      TIMESTAMPTZ DEFAULT NOW()
);
```

## MIGRATION 005 — SKILLS

```sql
-- Normalized skill catalog (not free text)
CREATE TABLE skills (
  id          SERIAL PRIMARY KEY,
  name        VARCHAR(100) UNIQUE NOT NULL,
  category_id INTEGER REFERENCES categories(id),
  is_active   BOOLEAN DEFAULT TRUE
);

-- User skills (many-to-many)
CREATE TABLE user_skills (
  user_id     UUID REFERENCES users(id) ON DELETE CASCADE,
  skill_id    INTEGER REFERENCES skills(id) ON DELETE CASCADE,
  PRIMARY KEY (user_id, skill_id)
);

-- Job required skills (many-to-many)
CREATE TABLE job_skills (
  job_id      UUID REFERENCES jobs(id) ON DELETE CASCADE,
  skill_id    INTEGER REFERENCES skills(id) ON DELETE CASCADE,
  PRIMARY KEY (job_id, skill_id)
);
```

## MIGRATION 006 — JOBS TABLE

```sql
CREATE TABLE jobs (
  id                      UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  provider_id             UUID NOT NULL REFERENCES users(id),
  title                   VARCHAR(200) NOT NULL,
  category_id             INTEGER REFERENCES categories(id),
  description             TEXT,
  work_type               VARCHAR(20) NOT NULL
                            CHECK (work_type IN ('hourly', 'daily', 'monthly')),

  -- ── HOURLY JOB FIELDS ──────────────────────────────────────────────────
  hourly_rate             INTEGER,                -- INR per hour
  total_hours             INTEGER,                -- Total hours needed
  shift_start_time        TIME,
  shift_end_time          TIME,
  same_day_payment        BOOLEAN DEFAULT FALSE,
  urgent_hiring           BOOLEAN DEFAULT FALSE,

  -- ── DAILY JOB FIELDS ───────────────────────────────────────────────────
  daily_wage              INTEGER,                -- INR per day
  num_working_days        INTEGER,
  food_included           BOOLEAN DEFAULT FALSE,
  accommodation_included  BOOLEAN DEFAULT FALSE,
  overtime_available      BOOLEAN DEFAULT FALSE,

  -- ── MONTHLY JOB FIELDS ─────────────────────────────────────────────────
  monthly_salary          INTEGER,                -- INR per month
  joining_date            DATE,
  experience_required     INTEGER DEFAULT 0,      -- Years of experience
  working_days_per_week   INTEGER,
  pf_esi_included         BOOLEAN DEFAULT FALSE,
  salary_negotiable       BOOLEAN DEFAULT FALSE,

  -- ── LOCATION ───────────────────────────────────────────────────────────
  latitude                DECIMAL(9,6),
  longitude               DECIMAL(9,6),
  address                 TEXT,
  city                    VARCHAR(100) NOT NULL,

  -- ── STATUS ─────────────────────────────────────────────────────────────
  status                  VARCHAR(20) DEFAULT 'active'
                            CHECK (status IN (
                              'draft', 'active', 'paused', 'filled', 'expired', 'removed'
                            )),
  is_urgent               BOOLEAN DEFAULT FALSE,
  is_verified             BOOLEAN DEFAULT FALSE,  -- Admin verified

  -- ── SCAM DETECTION ─────────────────────────────────────────────────────
  scam_score              DECIMAL(3,2) DEFAULT 0.00,
  flagged_count           INTEGER DEFAULT 0,

  -- ── FULL TEXT SEARCH ───────────────────────────────────────────────────
  search_vector           TSVECTOR,

  -- ── ANALYTICS ──────────────────────────────────────────────────────────
  views_count             INTEGER DEFAULT 0,
  applications_count      INTEGER DEFAULT 0,

  expires_at              TIMESTAMPTZ DEFAULT NOW() + INTERVAL '7 days',
  created_at              TIMESTAMPTZ DEFAULT NOW(),
  updated_at              TIMESTAMPTZ DEFAULT NOW()
);
```

## MIGRATION 007 — APPLICATIONS TABLE

```sql
CREATE TABLE applications (
  id                UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  job_id            UUID NOT NULL REFERENCES jobs(id),
  seeker_id         UUID NOT NULL REFERENCES users(id),

  status            VARCHAR(30) NOT NULL DEFAULT 'pending'
                      CHECK (status IN (
                        'pending',
                        'accepted',
                        'rejected',
                        'withdrawn',
                        'completed'
                      )),

  -- Tracking
  check_in_time     TIMESTAMPTZ,
  check_out_time    TIMESTAMPTZ,
  hours_worked      DECIMAL(5,2),

  employer_note     TEXT,                -- Employer reason for reject/accept

  applied_at        TIMESTAMPTZ DEFAULT NOW(),
  responded_at      TIMESTAMPTZ,
  completed_at      TIMESTAMPTZ,

  UNIQUE(job_id, seeker_id)             -- Prevent duplicate applications
);
```

## MIGRATION 008 — REVIEWS TABLE

```sql
CREATE TABLE reviews (
  id                UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  application_id    UUID REFERENCES applications(id),
  reviewer_id       UUID NOT NULL REFERENCES users(id),
  reviewee_id       UUID NOT NULL REFERENCES users(id),
  reviewer_role     VARCHAR(20) NOT NULL
                      CHECK (reviewer_role IN ('seeker', 'provider')),
  rating            INTEGER NOT NULL CHECK (rating BETWEEN 1 AND 5),
  comment           TEXT,
  created_at        TIMESTAMPTZ DEFAULT NOW(),

  UNIQUE(application_id, reviewer_id)   -- One review per person per job
);
```

## MIGRATION 010 — CHAT TABLES

```sql
-- One room per accepted application
CREATE TABLE chat_rooms (
  id                UUID PRIMARY KEY DEFAULT gen_random_uuid(),

-- Messages in each room
CREATE TABLE chat_messages (
  id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  room_id       UUID NOT NULL REFERENCES chat_rooms(id),
  sender_id     UUID NOT NULL REFERENCES users(id),
  message_type  VARCHAR(20) DEFAULT 'text'
                  CHECK (message_type IN ('text', 'system')),
  content       TEXT NOT NULL,
  is_read       BOOLEAN DEFAULT FALSE,
  created_at    TIMESTAMPTZ DEFAULT NOW()
```

## MIGRATION 011 — REPORTS TABLE

```sql
CREATE TABLE reports (
  id                UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  reporter_id       UUID NOT NULL REFERENCES users(id),
  reported_job_id   UUID REFERENCES jobs(id),
  reported_user_id  UUID REFERENCES users(id),
  reason            VARCHAR(100) NOT NULL,
  description       TEXT,
  status            VARCHAR(20) DEFAULT 'pending'
                      CHECK (status IN ('pending', 'reviewed', 'resolved', 'dismissed')),
  created_at        TIMESTAMPTZ DEFAULT NOW(),

  -- Must report either a job or a user
  CHECK (reported_job_id IS NOT NULL OR reported_user_id IS NOT NULL)
);
```

## MIGRATION 014 — TRIGGERS (AUTO-UPDATE)

```sql
-- Auto-update updated_at timestamps
CREATE OR REPLACE FUNCTION fn_update_updated_at()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_users_updated_at
  BEFORE UPDATE ON users
  FOR EACH ROW EXECUTE FUNCTION fn_update_updated_at();

CREATE TRIGGER trg_jobs_updated_at
  BEFORE UPDATE ON jobs
  FOR EACH ROW EXECUTE FUNCTION fn_update_updated_at();

-- Auto-build search_vector for Full Text Search
CREATE OR REPLACE FUNCTION fn_update_job_search_vector()
RETURNS TRIGGER AS $$
BEGIN
  NEW.search_vector := to_tsvector('english',
    COALESCE(NEW.title, '')       || ' ' ||
    COALESCE(NEW.description, '') || ' ' ||
    COALESCE(NEW.city, '')
  );
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_jobs_search_vector
  BEFORE INSERT OR UPDATE ON jobs
  FOR EACH ROW EXECUTE FUNCTION fn_update_job_search_vector();
```

## ENTITY RELATIONSHIP SUMMARY

```
users
  ├── seeker_profiles (1:1)
  ├── provider_profiles (1:1)
  ├── user_skills (1:many)
  ├── jobs (1:many, as provider)
  ├── applications (1:many, as seeker)
  ├── reviews_given (1:many, as reviewer)
  ├── reviews_received (1:many, as reviewee)
  ├── notifications (1:many)
  ├── chat_rooms_as_seeker (1:many)
  ├── chat_rooms_as_provider (1:many)
  ├── reports_submitted (1:many)
  └── refresh_tokens (1:many)

jobs
  ├── belongs to users (provider)
  ├── belongs to categories
  ├── job_skills (1:many)
  ├── applications (1:many)
  └── reports (1:many)

applications
  ├── belongs to jobs
  ├── belongs to users (seeker)
  ├── reviews (1:2 max, one from each side)
  └── chat_rooms (1:1)

chat_rooms
  ├── belongs to applications
  ├── belongs to users (seeker)
  ├── belongs to users (provider)
  └── chat_messages (1:many)
```
